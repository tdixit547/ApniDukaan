\section{Authentication APIs (2 Endpoints)}

This section covers user authentication including registration and login functionality.

% API 1: Register
\subsection{API 1: Register New User}

\begin{tcolorbox}[colback=postcolor!10, colframe=postcolor, title=POST]
\textbf{Endpoint:} \texttt{http://localhost:8080/api/auth/register}
\end{tcolorbox}

\textbf{Description:} Registers a new user account in the system.

\textbf{Request Body:}
\begin{lstlisting}
{
  "email": "raj.shamani@gmail.com",
  "password": "password123",
  "firstName": "Raj",
  "lastName": "Shamani",
  "phone": "9876543210"
}
\end{lstlisting}

\textbf{Expected Response (201 Created):}
\begin{lstlisting}
{
    "address": null,
    "email": "raj.shamani@gmail.com",
    "firstName": "Raj",
    "id": 12,
    "lastName": "Shamani",
    "name": "Raj Shamani",
    "phone": "9876543210",
    "phoneNumber": "9876543210",
    "role": "CUSTOMER"
}
\end{lstlisting}

\section{Appendix: Postman Testing Tips}

\subsection{Setting Up Requests}

\begin{enumerate}
    \item \textbf{For POST/PUT requests:}
    \begin{itemize}
        \item Go to the \textbf{Body} tab
        \item Select \textbf{raw}
        \item Choose \textbf{JSON} from the dropdown
        \item Paste the JSON body
    \end{itemize}
    
    \item \textbf{Replace placeholders:}
    \begin{itemize}
        \item \texttt{userId=1} $\rightarrow$ Use actual user ID from registration
        \item \texttt{/products/1} $\rightarrow$ Use actual product ID
    \end{itemize}
\end{enumerate}

\subsection{Recommended Test Flow}

\begin{enumerate}
    \item Register a new user (obtain userId)
    \item Login with the registered credentials
    \item Create categories (Admin)
    \item Create products (Admin)
    \item Add items to cart
    \item Complete checkout (create order)
    \item Add product reviews
    \item View orders and order details
\end{enumerate}
